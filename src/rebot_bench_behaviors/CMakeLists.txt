cmake_minimum_required(VERSION 3.22)
project(rebot_bench_behaviors)
find_package(moveit_pro_package REQUIRED)
moveit_pro_package()
set(THIS_PACKAGE_INCLUDE_DEPENDS moveit_pro_behavior_interface trajectory_msgs pluginlib cv_bridge sensor_msgs moveit_msgs geometry_msgs moveit_studio_vision_msgs ament_index_cpp)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d)
find_package(yaml-cpp REQUIRED)
find_package(moveit_pro_behavior REQUIRED)
foreach(package IN ITEMS ${THIS_PACKAGE_INCLUDE_DEPENDS})
  find_package(${package} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/record_rebot_trajectory.cpp src/anchor_trajectory_to_command.cpp src/clamp_robot_state_to_bounds.cpp
  src/hold_gripper_in_trajectory.cpp
  src/register_behaviors.cpp src/draw_place_spot.cpp src/fit_cube_to_mask.cpp src/is_cube_stacked.cpp src/is_joint_at_least.cpp src/load_motion_scales.cpp
  src/report_unreachable_cube.cpp src/plan_to_joint_goal_from_bounded_start.cpp)
target_link_libraries(${PROJECT_NAME} ${OpenCV_LIBS} yaml-cpp::yaml-cpp moveit_pro_behavior::plan_to_joint_goal)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} ${THIS_PACKAGE_INCLUDE_DEPENDS})
install(TARGETS ${PROJECT_NAME} EXPORT ${PROJECT_NAME}Targets LIBRARY DESTINATION lib ARCHIVE DESTINATION lib RUNTIME DESTINATION bin)
install(DIRECTORY include/ DESTINATION include)
pluginlib_export_plugin_description_file(moveit_pro_behavior_interface rebot_bench_behaviors_plugin_description.xml)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_gmock REQUIRED)
  ament_add_gmock(test_anchor_trajectory_to_command test/test_anchor_trajectory_to_command.cpp)
  ament_target_dependencies(test_anchor_trajectory_to_command ${THIS_PACKAGE_INCLUDE_DEPENDS})
  target_link_libraries(test_anchor_trajectory_to_command ${PROJECT_NAME})
  find_package(moveit_pro_test_utils REQUIRED)
  ament_add_gtest(test_fit_cube test/test_fit_cube.cpp)
  ament_target_dependencies(test_fit_cube moveit_pro_test_utils)
  target_link_libraries(test_fit_cube ${PROJECT_NAME})
  ament_add_gtest(test_draw_place_spot test/test_draw_place_spot.cpp)
  ament_target_dependencies(test_draw_place_spot moveit_pro_test_utils)
  target_link_libraries(test_draw_place_spot ${PROJECT_NAME})
  ament_add_gtest(test_hold_gripper test/test_hold_gripper.cpp)
  ament_target_dependencies(test_hold_gripper moveit_pro_test_utils)
  target_link_libraries(test_hold_gripper ${PROJECT_NAME})
  ament_add_gtest(test_stack_checks test/test_stack_checks.cpp)
  ament_target_dependencies(test_stack_checks moveit_pro_test_utils)
  target_link_libraries(test_stack_checks ${PROJECT_NAME})
endif()
ament_export_targets(${PROJECT_NAME}Targets HAS_LIBRARY_TARGET)
ament_export_dependencies(${THIS_PACKAGE_INCLUDE_DEPENDS} yaml-cpp moveit_pro_behavior)
ament_package()
